"""
LOAD — Charge les 8 tables brutes (CSV générés par extract.py) dans PostgreSQL.

Approche ELT : aucune transformation ici. La donnée est chargée telle quelle ;
le nettoyage et les jointures seront réalisés plus tard par dbt (SQL Pushdown).

L'ordre de chargement respecte les dépendances de clés étrangères :
  1. Tables de référence indépendantes : departements, annees_academiques, etudiants
  2. Tables dépendantes de niveau 1  : formations, enseignants
  3. Tables dépendantes de niveau 2  : cours
  4. Tables de faits (dépendent de tout le reste) : inscriptions, notes
"""
import os
import sys
from pathlib import Path

import pandas as pd
from sqlalchemy import create_engine, text
from sqlalchemy.engine import Engine

RAW_DIR = Path(__file__).resolve().parents[1] / "data" / "raw"

# Ordre de chargement : respecte strictement les dépendances de clés étrangères
LOAD_ORDER = [
    "departements",
    "annees_academiques",
    "etudiants",
    "formations",
    "enseignants",
    "cours",
    "inscriptions",
    "notes",
]


def get_engine() -> Engine:
    user = "etudiants_user"
    password = "etudiantscode212_passcode"
    host = "postgres"
    port = "5432"
    name = "etudiants_db"
    url = f"postgresql+psycopg2://{user}:{password}@{host}:{port}/{name}"
    return create_engine(url)


def truncate_all(engine: Engine) -> None:
    """Vide les tables dans l'ordre inverse (enfants d'abord) pour éviter
    les erreurs de contrainte FK lors d'un rechargement complet (idempotence)."""
    with engine.begin() as conn:
        for table in reversed(LOAD_ORDER):
            conn.execute(text(f"TRUNCATE TABLE {table} RESTART IDENTITY CASCADE;"))
    print("[LOAD] Tables vidées (TRUNCATE CASCADE) avant rechargement.")


def load_table(engine: Engine, table_name: str) -> int:
    csv_path = RAW_DIR / f"{table_name}.csv"
    if not csv_path.exists():
        raise FileNotFoundError(
            f"Fichier introuvable : {csv_path}. As-tu bien lancé extract.py avant load.py ?"
        )

    df = pd.read_csv(csv_path)
    df.to_sql(
        table_name,
        engine,
        if_exists="append",   # les tables existent déjà (créées par sql/init.sql)
        index=False,
        method="multi",
        chunksize=500,
    )
    return len(df)


def main() -> None:
    engine = get_engine()

    # Vérifie que la connexion fonctionne avant de commencer
    try:
        with engine.connect() as conn:
            conn.execute(text("SELECT 1"))
    except Exception as e:
        print(f"[LOAD] Impossible de se connecter à PostgreSQL : {e}")
        sys.exit(1)

    truncate_all(engine)

    total = 0
    for table_name in LOAD_ORDER:
        try:
            n = load_table(engine, table_name)
            total += n
            print(f"[LOAD] {table_name:<20} -> {n:>6} lignes chargées")
        except Exception as e:
            print(f"[LOAD] ERREUR sur la table '{table_name}' : {e}")
            sys.exit(1)

    print(f"[LOAD] Terminé — {total} lignes chargées au total dans {len(LOAD_ORDER)} tables.")


if __name__ == "__main__":
    main()

def style_header_row(ws, row, ncols):
    for c in range(1, ncols + 1):
        cell = ws.cell(row=row, column=c)
        cell.font = header_font
        cell.fill = header_fill
        cell.border = border
        cell.alignment = Alignment(vertical="center", wrap_text=True)


def write_table_sheet(wb, sheet_name, columns):
    ws = wb.create_sheet(sheet_name)
    ws.sheet_view.showGridLines = False

    # Titre
    ws["A1"] = f"Table : {sheet_name}"
    ws["A1"].font = title_font
    ws.merge_cells("A1:D1")

    header_row = 3
    for i, h in enumerate(HEADERS, start=1):
        ws.cell(row=header_row, column=i, value=h)
    style_header_row(ws, header_row, len(HEADERS))

    for r, (col, typ, key, desc) in enumerate(columns, start=header_row + 1):
        values = [col, typ, key, desc]
        for c, val in enumerate(values, start=1):
            cell = ws.cell(row=r, column=c, value=val)
            cell.font = normal_font
            cell.border = border
            cell.alignment = Alignment(vertical="center", wrap_text=(c == 4))
            if key == "PK":
                cell.fill = PatternFill(start_color="D9E1F2", end_color="D9E1F2", fill_type="solid")
            elif (r - header_row) % 2 == 0:
                cell.fill = PatternFill(start_color=LIGHT_BG, end_color=LIGHT_BG, fill_type="solid")

    for i, w in enumerate(COL_WIDTHS, start=1):
        ws.column_dimensions[get_column_letter(i)].width = w

    ws.freeze_panes = f"A{header_row + 1}"
    return ws


def write_summary_sheet(wb, tables):

# ==================================================================
def generate_formations(df_departements):

# ==================================================================
def generate_annees_academiques():

    id_ = 1
    for annee_debut in [2023, 2024, 2025]:
        libelle = f"{annee_debut}-{annee_debut + 1}"
        rows.append({
            "id_annee": id_, "libelle": libelle, "semestre": "S1",
            "date_debut": f"{annee_debut}-09-01", "date_fin": f"{annee_debut}-12-31",
        })
        id_ += 1
        rows.append({
            "id_annee": id_, "libelle": libelle, "semestre": "S2",
            "date_debut": f"{annee_debut + 1}-01-15", "date_fin": f"{annee_debut + 1}-06-30",
        })
        id_ += 1
    return pd.DataFrame(rows)


# ==================================================================
# 5. COURS (dépend de formations, enseignants, annees_academiques)
# ==================================================================
def generate_cours(df_formations, df_enseignants, df_annees):
    rows = []
    id_ = 1
    for _, formation in df_formations.iterrows():
        # même enseignant "pool" du département pour un peu de cohérence
        enseignants_dept = df_enseignants[df_enseignants["id_departement"] == formation["id_departement"]]
        if enseignants_dept.empty:
            enseignants_dept = df_enseignants
        for _, annee in df_annees.iterrows():
            nb_cours = random.randint(1, 3)
            for _ in range(nb_cours):
                rows.append({
                    "id_cours": id_,
                    "code_cours": f"{formation['nom_formation'][:3].upper()}{id_:04d}",
                    "intitule": fake.catch_phrase(),
                    "credits_ects": random.choice([3, 4, 5, 6]),
                    "id_formation": formation["id_formation"],
                    "id_enseignant": enseignants_dept.sample(1).iloc[0]["id_enseignant"],
                    "id_annee": annee["id_annee"],
                    "volume_horaire": random.choice([24, 30, 36, 48]),
                })
                id_ += 1
    return pd.DataFrame(rows)

            "nationalite": fake.country() if random.random() < 0.15 else "Marocaine",
            "email": fake.free_email(),
            "date_premiere_inscription": fake.date_between(start_date="-3y", end_date="-1y"),
            "statut": random.choices(STATUTS_ETUDIANT, weights=[0.75, 0.15, 0.10])[0],
        })
    return pd.DataFrame(rows)


# ==================================================================
# 7. INSCRIPTIONS (dépend de etudiants, formations, annees_academiques)
# ==================================================================
def generate_inscriptions(df_etudiants, df_formations, df_annees):
    rows = []
    id_ = 1
    annees_uniques = df_annees["libelle"].unique()
    for _, etudiant in df_etudiants.iterrows():
        formation = df_formations.sample(1).iloc[0]
        nb_annees_inscrit = random.randint(1, min(3, len(annees_uniques)))
        annees_choisies = random.sample(list(annees_uniques), nb_annees_inscrit)
        for lib in annees_choisies:
            # une inscription par semestre de cette année (S1 + S2)
            annees_du_libelle = df_annees[df_annees["libelle"] == lib]
            for _, annee in annees_du_libelle.iterrows():
                rows.append({
                    "id_inscription": id_,
                    "id_etudiant": etudiant["id_etudiant"],
                    "id_formation": formation["id_formation"],
                    "id_annee": annee["id_annee"],
                    "date_inscription": annee["date_debut"],
                    "statut_inscription": random.choices(STATUTS_INSCRIPTION, weights=[0.85, 0.1, 0.05])[0],
                    "mode": random.choices(MODES, weights=[0.8, 0.2])[0],
                })
                id_ += 1
    return pd.DataFrame(rows)


# ==================================================================
# 8. NOTES (dépend de etudiants, cours, annees_academiques, via inscriptions)
# ==================================================================
def generate_notes(df_inscriptions, df_cours):
    rows = []
    id_ = 1
    cours_par_formation_annee = df_cours.groupby(["id_formation", "id_annee"])

    for _, inscription in df_inscriptions.iterrows():
        try:
            cours_dispo = cours_par_formation_annee.get_group(
                (inscription["id_formation"], inscription["id_annee"])
            )
        except KeyError:
            continue

        for _, cours in cours_dispo.iterrows():
            n_evals = random.randint(1, 2)
            for _ in range(n_evals):
                # note réaliste : distribution centrée autour de 11-12/20
                note = round(min(20, max(0, random.gauss(11.5, 4.0))), 2)
                rows.append({
                    "id_note": id_,
                    "id_etudiant": inscription["id_etudiant"],
                    "id_cours": cours["id_cours"],
                    "id_annee": inscription["id_annee"],
                    "type_evaluation": random.choice(TYPES_EVAL),
                    "note": note,
                    "date_evaluation": fake.date_between(start_date="-2y", end_date="today"),
                })
                id_ += 1
    return pd.DataFrame(rows)


def main():
    RAW_DIR.mkdir(parents=True, exist_ok=True)

    print("[EXTRACT] Génération des entités de référence...")
    df_departements = generate_departements()
    df_formations = generate_formations(df_departements)
    df_enseignants = generate_enseignants(df_departements)
    df_annees = generate_annees_academiques()
    df_etudiants = generate_etudiants()

    print("[EXTRACT] Génération des entités dépendantes...")
    df_cours = generate_cours(df_formations, df_enseignants, df_annees)
    df_inscriptions = generate_inscriptions(df_etudiants, df_formations, df_annees)
    df_notes = generate_notes(df_inscriptions, df_cours)

    tables = {
        "departements": df_departements,
        "formations": df_formations,
        "enseignants": df_enseignants,
        "annees_academiques": df_annees,
        "cours": df_cours,
        "etudiants": df_etudiants,
        "inscriptions": df_inscriptions,
        "notes": df_notes,
    }

    for name, df in tables.items():
        path = RAW_DIR / f"{name}.csv"
        df.to_csv(path, index=False)
        print(f"[EXTRACT] {name:<20} -> {len(df):>6} lignes -> {path}")


if __name__ == "__main__":
    main()

with etudiants as (
    select * from {{ ref('stg_etudiants') }}
),
dedupliques as (
    select
        *,
        row_number() over (partition by id_etudiant order by id_etudiant) as rn
    from etudiants
    where id_etudiant is not null
),
enrichis as (
    select
        id_etudiant,
        matricule,
        nom,
        prenom,
        nom || ' ' || prenom                                        as nom_complet,
        date_naissance,
        date_part('year', age(current_date, date_naissance))::int   as age,
        sexe,
        nationalite,
        email,
        date_premiere_inscription,
        statut
    from dedupliques
    where rn = 1
)
select
    *,
    case
        when age < 20 then '18-19'
        when age < 23 then '20-22'
        when age < 26 then '23-25'
        else '26+'
    end as tranche_age
from enrichis

        row_number() over (partition by id_etudiant order by id_etudiant) as rn
    from etudiants
    where id_etudiant is not null
),
enrichis as (
    select
        id_etudiant,
        matricule,
        nom,
        prenom,
        nom || ' ' || prenom                                        as nom_complet,
        date_naissance,
        date_part('year', age(current_date, date_naissance))::int   as age,
        sexe,
        nationalite,
        email,
        date_premiere_inscription,
        statut
    from dedupliques
    where rn = 1
)
select
    *,
    case
        when age < 20 then '18-19'
        when age < 23 then '20-22'
        when age < 26 then '23-25'
        else '26+'
    end as tranche_age
from enrichis
  );

with etudiants as (
    select * from "etudiants_db"."public_staging"."stg_etudiants"
),
dedupliques as (
    select
        *,
        row_number() over (partition by id_etudiant order by id_etudiant) as rn
    from etudiants
    where id_etudiant is not null
),
enrichis as (
    select
        id_etudiant,
        matricule,
        nom,
        prenom,
        nom || ' ' || prenom                                        as nom_complet,
        date_naissance,
        date_part('year', age(current_date, date_naissance))::int   as age,
        sexe,
        nationalite,
        email,
        date_premiere_inscription,
        statut
    from dedupliques
    where rn = 1
)
select
    *,
    case
        when age < 20 then '18-19'
        when age < 23 then '20-22'
        when age < 26 then '23-25'
        else '26+'
    end as tranche_age
from enrichis

from __future__ import annotations
from datetime import datetime, timedelta
import sys
import os

from airflow.decorators import dag, task

sys.path.append("/opt/airflow/src")

default_args = {
    "owner": "data-team",
    "retries": 2,
    "retry_delay": timedelta(minutes=5),
}

@dag(
    dag_id="pipeline_academique_elt",
    description="Pipeline ELT complet : extract -> load -> dbt run -> dbt test",
    default_args=default_args,
    schedule_interval="@daily",
    start_date=datetime(2026, 1, 1),
    catchup=False,
    tags=["academique", "elt", "dbt"],
)
def pipeline_academique():

    @task
    def extract_task():
        from extract import main as extract_main
        extract_main()

    @task
    def load_task():
        from load import main as load_main
        load_main()

    @task
    def dbt_run_task():
        import subprocess
        import logging
        import os
        
        log = logging.getLogger("airflow.task")
        
        
        dbt_env = os.environ.copy()
        dbt_env["DBT_LOG_PATH"] = "/opt/airflow/dbt_target/logs"
        dbt_env["DBT_TARGET_PATH"] = "/opt/airflow/dbt_target/target"
        
        result = subprocess.run(
            ["dbt", "run", "--project-dir", "/opt/airflow/dbt_project", "--profiles-dir", "/opt/airflow/dbt_project"],
            env=dbt_env,
            capture_output=True, text=True,
        )
        
        log.info("---- DBT STDOUT ----\n%s", result.stdout)
        if result.stderr:
            log.error("---- DBT STDERR ----\n%s", result.stderr)
            
        if result.returncode != 0:
            raise RuntimeError(f"dbt run a échoué (code {result.returncode}). Chouf les logs l-foq bash t-3ref l-mochkil.")

    @task
    def dbt_test_task():
        import subprocess
        import logging
        import os
        
        log = logging.getLogger("airflow.task")
        
        
        dbt_env = os.environ.copy()
        dbt_env["DBT_LOG_PATH"] = "/opt/airflow/dbt_target/logs"
        dbt_env["DBT_TARGET_PATH"] = "/opt/airflow/dbt_target/target"
        
        result = subprocess.run(
            ["dbt", "test", "--project-dir", "/opt/airflow/dbt_project", "--profiles-dir", "/opt/airflow/dbt_project"],
            env=dbt_env,
            capture_output=True, text=True,
        )
        
        log.info("---- DBT STDOUT ----\n%s", result.stdout)
        if result.stderr:
            log.error("---- DBT STDERR ----\n%s", result.stderr)
            
        if result.returncode != 0:
            raise RuntimeError(f"dbt test a échoué (code {result.returncode}). Chouf les logs l-foq bash t-3ref l-mochkil.")

    e = extract_task()
    l = load_task()
    r = dbt_run_task()
    t = dbt_test_task()

    e >> l >> r >> t

pipeline_academique()

-- ============================
-- SCHEMA SOURCE (RAW) — Système d'Information Académique
-- ============================

CREATE TABLE IF NOT EXISTS departements (
    id_departement   SERIAL PRIMARY KEY,
    nom_departement  VARCHAR(150) NOT NULL,
    faculte          VARCHAR(150),
    responsable      VARCHAR(150)
);

CREATE TABLE IF NOT EXISTS formations (
    id_formation     SERIAL PRIMARY KEY,
    nom_formation    VARCHAR(150) NOT NULL,
    id_departement   INTEGER REFERENCES departements(id_departement),
    niveau           VARCHAR(30),
    type_diplome     VARCHAR(50),
    duree_annees     SMALLINT
);

CREATE TABLE IF NOT EXISTS enseignants (
    id_enseignant     SERIAL PRIMARY KEY,
    nom               VARCHAR(100) NOT NULL,
    prenom            VARCHAR(100) NOT NULL,
    grade             VARCHAR(50),
    id_departement    INTEGER REFERENCES departements(id_departement),
    email             VARCHAR(150),
    date_recrutement  DATE
);

CREATE TABLE IF NOT EXISTS etudiants (
    id_etudiant               SERIAL PRIMARY KEY,
    matricule                 VARCHAR(20) UNIQUE,
    nom                       VARCHAR(100) NOT NULL,
    prenom                    VARCHAR(100) NOT NULL,
    date_naissance            DATE,
    sexe                      CHAR(1),
    nationalite               VARCHAR(60),
    email                     VARCHAR(150),
    date_premiere_inscription DATE,
    statut                    VARCHAR(30)
);

CREATE TABLE IF NOT EXISTS annees_academiques (
    id_annee    SERIAL PRIMARY KEY,
    libelle     VARCHAR(20) NOT NULL,
    semestre    VARCHAR(5),
    date_debut  DATE,
    date_fin    DATE
);

CREATE TABLE IF NOT EXISTS cours (
    id_cours        SERIAL PRIMARY KEY,
    code_cours      VARCHAR(20),
    intitule        VARCHAR(150) NOT NULL,
    credits_ects    SMALLINT,
    id_formation    INTEGER REFERENCES formations(id_formation),
    id_enseignant   INTEGER REFERENCES enseignants(id_enseignant),
    id_annee        INTEGER REFERENCES annees_academiques(id_annee),
    volume_horaire  SMALLINT
);

CREATE TABLE IF NOT EXISTS inscriptions (
    id_inscription      SERIAL PRIMARY KEY,
    id_etudiant         INTEGER REFERENCES etudiants(id_etudiant),
    id_formation        INTEGER REFERENCES formations(id_formation),
    id_annee            INTEGER REFERENCES annees_academiques(id_annee),
    date_inscription    DATE,
    statut_inscription  VARCHAR(30),
    mode                VARCHAR(20)
);

CREATE TABLE IF NOT EXISTS notes (
    id_note          SERIAL PRIMARY KEY,
    id_etudiant      INTEGER REFERENCES etudiants(id_etudiant),
    id_cours         INTEGER REFERENCES cours(id_cours),
    id_annee         INTEGER REFERENCES annees_academiques(id_annee),
    type_evaluation  VARCHAR(30),
    note             NUMERIC(4,2) CHECK (note BETWEEN 0 AND 20),
    date_evaluation  DATE
);

-- Index utiles pour les futures jointures et filtres BI
CREATE INDEX IF NOT EXISTS idx_notes_etudiant ON notes(id_etudiant);
CREATE INDEX IF NOT EXISTS idx_notes_cours ON notes(id_cours);
CREATE INDEX IF NOT EXISTS idx_inscriptions_etudiant ON inscriptions(id_etudiant);
CREATE INDEX IF NOT EXISTS idx_cours_formation ON cours(id_formation);
